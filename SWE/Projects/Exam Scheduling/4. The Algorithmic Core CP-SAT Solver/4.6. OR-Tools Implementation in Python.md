# 4.6. OR-Tools Implementation in Python

> [!abstract] What this note teaches you
> The complete V2 `ExamScheduler` class in Python using Google OR-Tools CP-SAT. This is the actual code that replaces V1's `generate_schedule.sql` PL/pgSQL function. Variables, hard constraints, objective, and solver invocation — all in one place.

## The class skeleton

```python
# scheduler/cp_sat_solver.py
from ortools.sat.python import cp_model
from typing import Dict, List, Tuple

class ExamScheduler:
    def __init__(self, data: 'ScheduleData'):
        self.data = data
        self.model = cp_model.CpModel()
        self.x: Dict[Tuple[int, int], cp_model.IntVar] = {}  # (module, creneau)
        self.y: Dict[Tuple[int, int, int], cp_model.IntVar] = {}  # (module, room, creneau)
        self.z: Dict[Tuple[int, int, int], cp_model.IntVar] = {}  # (module, prof, creneau)

    def solve(self, timeout_seconds: float = 30.0) -> 'ScheduleSolution':
        self.build_variables()
        self.add_hard_constraints()
        self.add_objective()
        solver = cp_model.CpSolver()
        solver.parameters.max_time_in_seconds = timeout_seconds
        solver.parameters.num_search_workers = 8
        solver.parameters.log_search_progress = True
        status = solver.Solve(self.model)
        return ScheduleSolution.from_solver(solver, status, self.x, self.y, self.z)
```

## Building variables

```python
def build_variables(self):
    for m in self.data.modules:
        for c in self.data.creneaux_for_module(m):
            # x[m, c] = 1 if module m scheduled at creneau c
            self.x[m.id, c.id] = self.model.NewBoolVar(f'x_{m.id}_{c.id}')

            # y[m, r, c] for each feasible room
            for r in self.data.rooms_for_module(m):
                # Skip infeasible rooms upfront (capacity, type, equipment)
                if r.capacity < m.enrollment and r.type != 'salle':
                    continue  # will fail constraint 4
                if m.type_lieu_requis != 'any' and r.type != m.type_lieu_requis:
                    continue  # will fail constraint 13
                if not set(m.equipements_requis).issubset(set(r.equipements)):
                    continue  # will fail constraint 14
                self.y[m.id, r.id, c.id] = self.model.NewBoolVar(
                    f'y_{m.id}_{r.id}_{c.id}')

            # z[m, p, c] for each feasible professor
            for p in self.data.profs_for_module(m):
                self.z[m.id, p.id, c.id] = self.model.NewBoolVar(
                    f'z_{m.id}_{p.id}_{c.id}')
```

The pre-filtering (skipping infeasible rooms and profs) reduces variable count by 80–90% — critical for solver performance.

## Adding all 16 hard constraints

```python
def add_hard_constraints(self):
    # Constraint 1: Each module scheduled at most once
    for m in self.data.modules:
        self.model.Add(
            sum(self.x[m.id, c.id] for c in self.data.creneaux_for_module(m)
                if (m.id, c.id) in self.x) <= 1
        )

    # Constraint 2: Conflict modules can't share creneau
    for (m1, m2) in self.data.conflicts:
        for c in self.data.creneaux:
            if (m1, c.id) in self.x and (m2, c.id) in self.x:
                self.model.Add(self.x[m1, c.id] + self.x[m2, c.id] <= 1)

    # Constraint 3: Student max exams per day
    for student, modules in self.data.student_modules.items():
        for day, day_creneaux in self.data.creneaux_by_day.items():
            terms = [self.x[m_id, c.id]
                     for m_id in modules
                     for c in day_creneaux
                     if (m_id, c.id) in self.x]
            if terms:
                self.model.Add(sum(terms) <= self.data.max_exams_per_day)

    # Constraint 4 & 5: Room capacity + one room per module (or multiple for splits)
    for m in self.data.modules:
        for c in self.data.creneaux_for_module(m):
            if (m.id, c.id) not in self.x:
                continue
            rooms = [r for r in self.data.rooms_for_module(m)
                     if (m.id, r.id, c.id) in self.y]
            if rooms:
                # y sums to x
                self.model.Add(
                    sum(self.y[m.id, r.id, c.id] for r in rooms) == self.x[m.id, c.id]
                )
                # Capacity: sum of capacities >= enrollment
                self.model.Add(
                    sum(self.y[m.id, r.id, c.id] * r.capacity for r in rooms) >=
                    m.enrollment * self.x[m.id, c.id]
                )

    # Constraint 6: One professor per scheduled module
    for m in self.data.modules:
        for c in self.data.creneaux_for_module(m):
            if (m.id, c.id) not in self.x:
                continue
            profs = [p for p in self.data.profs_for_module(m)
                     if (m.id, p.id, c.id) in self.z]
            if profs:
                self.model.Add(
                    sum(self.z[m.id, p.id, c.id] for p in profs) >= self.x[m.id, c.id]
                )

    # Constraint 7: Room not double-booked
    for r in self.data.rooms:
        for c in self.data.creneaux:
            terms = [self.y[m.id, r.id, c.id]
                     for m in self.data.modules_at_creneau(c)
                     if (m.id, r.id, c.id) in self.y]
            if terms:
                self.model.Add(sum(terms) <= 1)

    # Constraint 8: Professor not double-booked
    for p in self.data.professors:
        for c in self.data.creneaux:
            terms = [self.z[m.id, p.id, c.id]
                     for m in self.data.modules_at_creneau(c)
                     if (m.id, p.id, c.id) in self.z]
            if terms:
                self.model.Add(sum(terms) <= 1)

    # Constraint 9: Professor daily limit
    for p in self.data.professors:
        for day, day_creneaux in self.data.creneaux_by_day.items():
            terms = [self.z[m.id, p.id, c.id]
                     for m in self.data.modules
                     for c in day_creneaux
                     if (m.id, p.id, c.id) in self.z]
            if terms:
                self.model.Add(sum(terms) <= p.max_surveillances_jour)

    # Constraint 10: Professor session limit
    for p in self.data.professors:
        terms = [self.z[m.id, p.id, c.id]
                 for m in self.data.modules
                 for c in self.data.creneaux
                 if (m.id, p.id, c.id) in self.z]
        if terms:
            self.model.Add(sum(terms) <= p.max_surveillances_session)

    # Constraint 11: Professor availability
    for p in self.data.professors:
        for unavail in p.indisponibilites:
            for c in self.data.creneaux_overlapping(unavail):
                for m in self.data.modules:
                    if (m.id, p.id, c.id) in self.z:
                        self.model.Add(self.z[m.id, p.id, c.id] == 0)

    # Constraint 12: Room availability
    for r in self.data.rooms:
        for unavail in r.indisponibilites:
            for c in self.data.creneaux_overlapping(unavail):
                for m in self.data.modules:
                    if (m.id, r.id, c.id) in self.y:
                        self.model.Add(self.y[m.id, r.id, c.id] == 0)

    # Constraint 15: TD rooms ≤ 20 students
    for m in self.data.modules:
        for c in self.data.creneaux_for_module(m):
            for r in self.data.rooms_for_module(m):
                if r.type == 'salle' and m.enrollment > 20:
                    if (m.id, r.id, c.id) in self.y:
                        self.model.Add(self.y[m.id, r.id, c.id] == 0)

    # Constraint 16: Duration fits
    for m in self.data.modules:
        for c in self.data.creneaux_for_module(m):
            if c.duration_minutes < m.duree_examen_minutes:
                if (m.id, c.id) in self.x:
                    self.model.Add(self.x[m.id, c.id] == 0)
```

(Constraints 13 and 14 — room type and equipment — are handled in `build_variables` by not creating the `y` variables.)

## Adding the objective

```python
def add_objective(self):
    # W1: Unscheduled penalty (weight 1000)
    unscheduled = []
    for m in self.data.modules:
        terms = [self.x[m.id, c.id]
                 for c in self.data.creneaux_for_module(m)
                 if (m.id, c.id) in self.x]
        if terms:
            unscheduled.append(1 - sum(terms))
        else:
            unscheduled.append(1)

    # W2: Fairness (weight 100) — variance of professor workloads
    fairness = self._fairness_term()

    # W3: Cross-dept proctoring (weight 10)
    cross_dept = []
    for m in self.data.modules:
        for p in self.data.professors:
            if p.departement_id != m.departement_id:
                for c in self.data.creneaux:
                    if (m.id, p.id, c.id) in self.z:
                        cross_dept.append(self.z[m.id, p.id, c.id])

    # W5: Weekend avoidance (weight 3)
    weekend = []
    for c in self.data.creneaux:
        if c.jour in (6, 7):  # Sat, Sun
            for m in self.data.modules:
                if (m.id, c.id) in self.x:
                    weekend.append(self.x[m.id, c.id])

    # W6: Room mismatch (weight 1)
    room_mismatch = []
    for m in self.data.modules:
        for c in self.data.creneaux_for_module(m):
            for r in self.data.rooms_for_module(m):
                if (m.id, r.id, c.id) in self.y:
                    waste = max(0, r.capacity - m.enrollment)
                    room_mismatch.append(waste * self.y[m.id, r.id, c.id])

    self.model.Minimize(
        1000 * sum(unscheduled) +
        100 * fairness +
        10 * sum(cross_dept) +
        3 * sum(weekend) +
        1 * sum(room_mismatch)
    )

def _fairness_term(self):
    """Approximate sum of (workload - avg)^2 using piecewise linear."""
    workloads = []
    for p in self.data.professors:
        terms = [self.z[m.id, p.id, c.id]
                 for m in self.data.modules
                 for c in self.data.creneaux
                 if (m.id, p.id, c.id) in self.z]
        if terms:
            w = self.model.NewIntVar(0, len(terms), f'workload_{p.id}')
            self.model.Add(w == sum(terms))
            workloads.append((p.id, w))

    if not workloads:
        return 0

    # Use absolute deviation as a linear proxy for variance
    # (sum of |w - avg| is minimized when w are equal)
    # For exact quadratic, use AddMultiplicationEquality (slower)
    total = self.model.NewIntVar(0, sum(w[1].Max() for w in workloads), 'total_workload')
    self.model.Add(total == sum(w[1] for w in workloads))
    avg = self.model.NewIntVar(0, max(w[1].Max() for w in workloads), 'avg_workload')
    self.model.AddDivisionEquality(avg, total, len(workloads))

    deviations = []
    for pid, w in workloads:
        dev = self.model.NewIntVar(0, w.Max(), f'dev_{pid}')
        self.model.AddAbsEquality(dev, w - avg)
        deviations.append(dev)

    return sum(deviations)
```

## The solver invocation

```python
def solve(self, timeout_seconds: float = 30.0) -> 'ScheduleSolution':
    self.build_variables()
    self.add_hard_constraints()
    self.add_objective()

    solver = cp_model.CpSolver()
    solver.parameters.max_time_in_seconds = timeout_seconds
    solver.parameters.num_search_workers = 8
    solver.parameters.log_search_progress = True

    status = solver.Solve(self.model)

    return ScheduleSolution.from_solver(
        solver=solver,
        status=status,
        x=self.x,
        y=self.y,
        z=self.z,
        modules=self.data.modules,
    )
```

## The `ScheduleSolution` wrapper

```python
class ScheduleSolution:
    def __init__(self, status, objective_value, assignments, scheduled_count,
                 unscheduled_count, duration_ms):
        self.status = status  # 'OPTIMAL', 'FEASIBLE', 'INFEASIBLE', 'UNKNOWN'
        self.objective_value = objective_value
        self.assignments = assignments  # list of {module_id, creneau_id, room_ids, prof_ids}
        self.scheduled_count = scheduled_count
        self.unscheduled_count = unscheduled_count
        self.duration_ms = duration_ms

    @property
    def is_feasible(self):
        return self.status in ('OPTIMAL', 'FEASIBLE')

    @classmethod
    def from_solver(cls, solver, status, x, y, z, modules):
        status_name = {
            cp_model.OPTIMAL: 'OPTIMAL',
            cp_model.FEASIBLE: 'FEASIBLE',
            cp_model.INFEASIBLE: 'INFEASIBLE',
            cp_model.UNKNOWN: 'UNKNOWN',
        }[status]

        assignments = []
        scheduled = 0
        for m in modules:
            placed = False
            for c_id in [c.id for c in m.creneaux]:
                if (m.id, c_id) in x and solver.Value(x[m.id, c_id]) == 1:
                    room_ids = [r_id for (mid, r_id, cid), var in y.items()
                                if mid == m.id and cid == c_id and solver.Value(var) == 1]
                    prof_ids = [p_id for (mid, p_id, cid), var in z.items()
                                if mid == m.id and cid == c_id and solver.Value(var) == 1]
                    assignments.append({
                        'module_id': m.id,
                        'creneau_id': c_id,
                        'room_ids': room_ids,
                        'prof_ids': prof_ids,
                    })
                    scheduled += 1
                    placed = True
                    break
            if not placed:
                assignments.append({
                    'module_id': m.id,
                    'creneau_id': None,
                    'room_ids': [],
                    'prof_ids': [],
                })

        return cls(
            status=status_name,
            objective_value=solver.ObjectiveValue() if status in (cp_model.OPTIMAL, cp_model.FEASIBLE) else None,
            assignments=assignments,
            scheduled_count=scheduled,
            unscheduled_count=len(modules) - scheduled,
            duration_ms=int(solver.WallTime() * 1000),
        )
```

## Common junior developer mistakes

- **Creating variables for infeasible combinations.** Pre-filter in `build_variables`. A 400-student exam should not have `y` variables for 20-seat rooms.
- **Not using `log_search_progress = True` during development.** You'll have no idea what the solver is doing.
- **Setting `num_search_workers = 1`.** CP-SAT parallelizes well. Use 8 (or however many cores you have).
- **Forgetting to handle the `UNKNOWN` status.** This means the solver timed out without finding any solution — different from `INFEASIBLE` (proof that no solution exists).
- **Returning raw CP-SAT objects.** Wrap the result in a `ScheduleSolution` so the Laravel side can deserialize predictably.
