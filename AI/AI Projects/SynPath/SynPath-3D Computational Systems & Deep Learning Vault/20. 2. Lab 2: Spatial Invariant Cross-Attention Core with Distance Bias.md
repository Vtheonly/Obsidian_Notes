### 1. Lab Objective & Mathematical Blueprint
Implement the complete cross-attention mechanism connecting the reacting handle state to the pocket node representations. 

The module incorporates continuous spatial distance penalties and goal-directed TIF anchor queries, returning an invariant latent planning vector $\mathbf{z}_t \in \mathbb{R}^{256}$.

### 2. Production PyTorch Code

```python
import math
import torch
import torch.nn as nn
import torch.nn.functional as F

class SpatialCrossAttentionCore(nn.Module):
    def __init__(self, d_model=256, num_heads=8, num_anchors=5):
        super().__init__()
        self.d_model = d_model
        self.num_heads = num_heads
        self.d_k = d_model // num_heads # 256 // 8 = 32
        
        # Handle & Global Feature Projections
        # 64 chemical handle + 4 global features = 68
        self.query_proj = nn.Linear(68, d_model)
        self.anchor_fuse = nn.Linear(d_model + 32, d_model)
        self.query_norm = nn.LayerNorm(d_model)
        
        # Multi-Head Attention Projections
        self.W_q = nn.Linear(d_model, d_model)
        self.W_k = nn.Linear(d_model, d_model)
        self.W_v = nn.Linear(d_model, d_model)
        self.out_proj = nn.Linear(d_model, d_model)
        
        # Spatial Distance MLP Bias
        self.dist_rbf = nn.Sequential(
            nn.Linear(1, 32),
            nn.SiLU(),
            nn.Linear(32, d_model)
        )
        self.gamma_dist = nn.Parameter(torch.tensor(0.10))

    def forward(self, handle_chem, global_feats, handle_pos, pocket_nodes, pocket_pos, tif_anchors=None):
        """
        handle_chem:   [B, 64]
        global_feats:  [B, 4]
        handle_pos:    [B, 3]
        pocket_nodes:  [B, N_p, 256]
        pocket_pos:    [B, N_p, 3]
        tif_anchors:   dict containing 'coords' [B, M, 3] and 'tokens' [B, M, 256]
        """
        B, N_p, _ = pocket_nodes.shape
        
        # 1. Base Query Construction
        q_base = self.query_proj(torch.cat([handle_chem, global_feats], dim=-1)) # [B, 256]
        
        # 2. Inject Unfulfilled TIF Anchors (Distal Goal Guidance)
        if tif_anchors is not None:
            anchor_coords = tif_anchors['coords'] # [B, M, 3]
            anchor_tokens = tif_anchors['tokens'] # [B, M, 256]
            # Distances from handle to all M anchors
            dist_to_anchors = torch.norm(anchor_coords - handle_pos.unsqueeze(1), dim=-1, keepdim=True) # [B, M, 1]
            anchor_rbf = torch.exp(-0.5 * (dist_to_anchors ** 2)) # [B, M, 1] - simplified Gaussian
            
            # Pool anchor context into query
            anchor_context = self.anchor_fuse(torch.cat([anchor_tokens, anchor_rbf.expand(-1, -1, 32)], dim=-1))
            q_base = q_base + anchor_context.mean(dim=1) # [B, 256]
            
        q_norm = self.query_norm(q_base).unsqueeze(1) # [B, 1, 256]
        
        # 3. Shape Projections for Multi-Head Attention
        Q = self.W_q(q_norm).view(B, 1, self.num_heads, self.d_k).transpose(1, 2)     # [B, H, 1, d_k]
        K = self.W_k(pocket_nodes).view(B, N_p, self.num_heads, self.d_k).transpose(1, 2) # [B, H, N_p, d_k]
        V = self.W_v(pocket_nodes).view(B, N_p, self.num_heads, self.d_k).transpose(1, 2) # [B, H, N_p, d_k]
        
        # 4. Compute Invariant Point Distances for Spatial Bias
        # dist_matrix: [B, 1, N_p]
        dists = torch.norm(pocket_pos - handle_pos.unsqueeze(1), dim=-1, keepdim=True).transpose(1, 2)
        spatial_penalty = self.gamma_dist * (dists ** 2).unsqueeze(1) # [B, 1, 1, N_p]
        
        # 5. Scaled Dot-Product with Continuous Spatial Attenuation
        scores = torch.matmul(Q, K.transpose(-2, -1)) / math.sqrt(self.d_k) # [B, H, 1, N_p]
        scores = scores - spatial_penalty # Attenuate distant pocket residues
        
        attn_weights = torch.softmax(scores, dim=-1) # [B, H, 1, N_p]
        
        # 6. Aggregate Value Embeddings
        context = torch.matmul(attn_weights, V) # [B, H, 1, d_k]
        context = context.transpose(1, 2).contiguous().view(B, self.d_model) # [B, 256]
        
        z_t = self.out_proj(context) # [B, 256]
        return z_t
```
