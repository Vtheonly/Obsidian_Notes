### 1. Lab Objective & Mathematical Blueprint
Implement the complete input featurization and radial expansion module for the 6-layer Equiformer-Light backbone.

The module takes raw atomic numbers ($Z$), physiological formal charges ($q$), continuous solvent-accessible surface areas ($\text{SASA}$), and continuous ENSF water potentials ($\Delta \mu$). It expands interatomic distances using $\mathcal{C}^2$-continuous Bessel basis functions and returns initial scalar node features $\mathbf{s}_i^{(0)} \in \mathbb{R}^{256}$ and vector features $\vec{v}_i^{(0)} \in \mathbb{R}^{3 \times 64}$.

### 2. Production PyTorch Code

```python
import math
import torch
import torch.nn as nn
import torch.nn.functional as F

class PolynomialCutoff(nn.Module):
    """Smooth C2-continuous polynomial cutoff envelope that decays to zero at r_cut."""
    def __init__(self, r_cut=6.0):
        super().__init__()
        self.r_cut = float(r_cut)

    def forward(self, dists):
        # dists: arbitrary shape tensor of positive Euclidean distances
        x = dists / self.r_cut
        envelope = 1.0 - 6.0*(x**5) + 15.0*(x**4) - 10.0*(x**3)
        return torch.where(dists < self.r_cut, envelope, torch.zeros_like(envelope))

class SphericalBesselRBF(nn.Module):
    """Spherical Bessel basis functions satisfying Helmholtz boundary conditions."""
    def __init__(self, num_rbf=32, r_cut=6.0):
        super().__init__()
        self.num_rbf = int(num_rbf)
        self.r_cut = float(r_cut)
        # Frequencies are initialized to match the roots of the 0th spherical Bessel function
        frequencies = torch.arange(1, num_rbf + 1, dtype=torch.float32) * (math.pi / self.r_cut)
        self.frequencies = nn.Parameter(frequencies)
        self.cutoff = PolynomialCutoff(r_cut)

    def forward(self, dists):
        # dists: [Num_Edges]
        d_expanded = dists.unsqueeze(-1) # [Num_Edges, 1]
        freq = self.frequencies.unsqueeze(0) # [1, Num_RBF]
        
        # phi_n(d) = sqrt(2/r_cut) * sin(n*pi*d / r_cut) / d
        norm_factor = math.sqrt(2.0 / self.r_cut)
        d_scaled = d_expanded * freq
        bessel = norm_factor * torch.sin(d_scaled) / d_expanded.clamp(min=1e-6)
        
        # Apply C2 polynomial cutoff envelope
        return bessel * self.cutoff(d_expanded) # [Num_Edges, Num_RBF]

class PocketNodeFeaturizer(nn.Module):
    """Fuses discrete atomic features with continuous physical-chemistry channels."""
    def __init__(self, d_model=256, max_z=100):
        super().__init__()
        self.d_model = d_model
        
        # 1. Discrete Element Embeddings
        self.element_embed = nn.Embedding(max_z, 128)
        
        # 2. Continuous Physicochemical Projections
        self.charge_mlp = nn.Sequential(
            nn.Linear(1, 32),
            nn.SiLU(),
            nn.Linear(32, 32)
        )
        self.sasa_mlp = nn.Sequential(
            nn.Linear(1, 32),
            nn.SiLU(),
            nn.Linear(32, 32)
        )
        self.ensf_mlp = nn.Sequential(
            nn.Linear(1, 64),
            nn.SiLU(),
            nn.Linear(64, 64)
        )
        
        # 3. Output Fusion LayerNorm
        self.fusion = nn.Linear(128 + 32 + 32 + 64, d_model)
        self.layer_norm = nn.LayerNorm(d_model)

    def forward(self, z, charges, sasa, ensf_potential):
        # z: [N], charges: [N], sasa: [N], ensf_potential: [N]
        e_z = self.element_embed(z)
        e_q = self.charge_mlp(charges.unsqueeze(-1))
        e_sasa = self.sasa_mlp(sasa.unsqueeze(-1))
        e_ensf = self.ensf_mlp(ensf_potential.unsqueeze(-1))
        
        # Concatenate along feature dimension
        fused = torch.cat([e_z, e_q, e_sasa, e_ensf], dim=-1) # [N, 256]
        s_0 = self.layer_norm(self.fusion(fused))             # [N, 256]
        
        # Vector features initialize to strict zero (l=1 irreps)
        v_0 = torch.zeros(z.size(0), 3, 64, device=z.device, dtype=s_0.dtype)
        return s_0, v_0
```
