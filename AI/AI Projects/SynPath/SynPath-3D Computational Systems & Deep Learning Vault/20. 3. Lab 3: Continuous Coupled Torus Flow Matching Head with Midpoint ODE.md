### 1. Lab Objective & Mathematical Blueprint
Implement the complete continuous-time Riemannian Flow Matching head on the flat torus $\mathbb{T}^K$. 

The module provides both training (evaluating wrapped velocity loss) and inference (integrating the 4-step Midpoint ODE solver), returning physically valid dihedral angles $\vec{\phi}^* \in [-\pi, \pi)^K$.

### 2. Production PyTorch Code

```python
import math
import torch
import torch.nn as nn
import torch.nn.functional as F

class TorusVelocityField(nn.Module):
    """Neural vector field predicting continuous tangent angular velocities on T^K."""
    def __init__(self, d_model=256, max_k=6):
        super().__init__()
        self.d_model = d_model
        self.max_k = max_k
        
        # Time embedding MLP
        self.time_mlp = nn.Sequential(
            nn.Linear(16, 64),
            nn.SiLU(),
            nn.Linear(64, 64)
        )
        
        # Joint residual MLP
        # Input: z_t (256) + E_B (256) + time (64) + circular angles [sin, cos] (2 * max_k = 12) = 588
        self.input_layer = nn.Linear(d_model + d_model + 64 + (2 * max_k), 256)
        
        self.res_block = nn.Sequential(
            nn.SiLU(),
            nn.Linear(256, 256),
            nn.SiLU(),
            nn.Linear(256, 256)
        )
        self.vel_out = nn.Linear(256, max_k)

    def _embed_time(self, t):
        # t: [B]
        half_dim = 8
        emb = math.log(10000) / (half_dim - 1)
        freq = torch.exp(torch.arange(half_dim, device=t.device, dtype=torch.float32) * -emb)
        args = t.unsqueeze(-1) * freq.unsqueeze(0)
        return self.time_mlp(torch.cat([torch.sin(args), torch.cos(args)], dim=-1)) # [B, 64]

    def forward(self, phi, t, z_t, E_B, k_mask=None):
        """
        phi:    [B, max_k] (Current dihedrals in [-pi, pi))
        t:      [B]        (Flow time in [0, 1])
        z_t:    [B, 256]   (Planning state)
        E_B:    [B, 256]   (Chosen synthon embedding)
        k_mask: [B, max_k] (Binary mask for active rotatable bonds)
        """
        B = phi.shape[0]
        e_time = self._embed_time(t) # [B, 64]
        
        # Expand circular dihedrals to avoid branch cuts: [B, 2 * max_k]
        phi_circ = torch.cat([torch.sin(phi), torch.cos(phi)], dim=-1)
        
        h = self.input_layer(torch.cat([z_t, E_B, e_time, phi_circ], dim=-1))
        h = h + self.res_block(h)
        v_pred = self.vel_out(h) # [B, max_k]
        
        if k_mask is not None:
            v_pred = v_pred * k_mask
        return v_pred

class TorusFlowMatchingModule(nn.Module):
    """Complete module providing both training loss evaluation and ODE sampling."""
    def __init__(self, d_model=256, max_k=6):
        super().__init__()
        self.max_k = max_k
        self.velocity_net = TorusVelocityField(d_model=d_model, max_k=max_k)

    def compute_train_loss(self, z_t, E_B, phi_1_target, k_mask):
        """
        Supervised Flow Matching loss evaluation on T^K.
        phi_1_target: [B, max_k] (Ground truth crystal dihedrals)
        """
        B = z_t.shape[0]
        
        # 1. Sample uniform noise on flat torus T^K
        phi_0 = torch.empty_like(phi_1_target).uniform_(-math.pi, math.pi)
        
        # 2. Sample random flow time t in [0, 1]
        t = torch.rand(B, device=z_t.device)
        
        # 3. Compute shortest arc target velocity vector
        diff = phi_1_target - phi_0
        u_star = torch.remainder(diff + math.pi, 2 * math.pi) - math.pi # [B, max_k]
        
        # 4. Interpolate state along circular geodesic
        phi_t = torch.remainder(phi_0 + t.unsqueeze(-1) * u_star + math.pi, 2 * math.pi) - math.pi
        
        # 5. Predict velocity field
        v_pred = self.velocity_net(phi_t, t, z_t, E_B, k_mask)
        
        # 6. Evaluate wrapped squared error metric
        vel_err = torch.abs(v_pred - u_star)
        wrapped_err = torch.minimum(vel_err, 2 * math.pi - vel_err)
        
        # Mask out inactive dihedrals and compute mean loss
        loss = (wrapped_err ** 2) * k_mask
        return loss.sum() / k_mask.sum().clamp(min=1.0)

    @torch.no_grad()
    def sample_dihedrals(self, z_t, E_B, k_mask, steps=4):
        """
        Inference ODE integration using a 4-step Midpoint solver on T^K.
        """
        B = z_t.shape[0]
        dt = 1.0 / steps
        
        # Base distribution: Uniform noise on T^K
        phi = torch.empty(B, self.max_k, device=z_t.device).uniform_(-math.pi, math.pi)
        
        for step in range(steps):
            t_val = step * dt
            t = torch.full((B,), t_val, device=z_t.device)
            
            # Step 1: Velocity at start of step
            k1 = self.velocity_net(phi, t, z_t, E_B, k_mask)
            
            # Step 2: State at midpoint
            phi_mid = torch.remainder(phi + 0.5 * dt * k1 + math.pi, 2 * math.pi) - math.pi
            t_mid = t + 0.5 * dt
            
            # Step 3: Velocity at midpoint
            k2 = self.velocity_net(phi_mid, t_mid, z_t, E_B, k_mask)
            
            # Step 4: Full step update using midpoint velocity
            phi = torch.remainder(phi + dt * k2 + math.pi, 2 * math.pi) - math.pi
            
        return phi * k_mask # [B, max_k] Final predicted physical dihedrals
```
