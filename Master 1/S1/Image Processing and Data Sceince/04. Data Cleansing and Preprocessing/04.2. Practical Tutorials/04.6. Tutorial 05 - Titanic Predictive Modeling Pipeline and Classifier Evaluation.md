# 04.6. Tutorial 05 - Titanic Predictive Modeling Pipeline and Classifier Evaluation

## 1. Overview
This tutorial builds an end-to-end machine learning pipeline on the Titanic dataset, covering data cleaning, feature engineering, categorical encoding, feature scaling, and model evaluation with Logistic Regression and K-Nearest Neighbors (KNN).

## 2. Complete Code Implementation

```python
import seaborn as sns
import pandas as pd
import numpy as np
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler
from sklearn.impute import SimpleImputer
from sklearn.linear_model import LogisticRegression
from sklearn.neighbors import KNeighborsClassifier
from sklearn.metrics import accuracy_score, classification_report, confusion_matrix

# 1. Ingestion and Redundant Column Pruning
raw_df = sns.load_dataset('titanic')
# Drop redundant columns: deck (high missingness), alive (target alias), embark_town (duplicate)
df = raw_df.drop(columns=['deck', 'alive', 'embark_town', 'who', 'class'])

# 2. Missing Value Imputation
age_imputer = SimpleImputer(strategy='median')
df['age'] = age_imputer.fit_transform(df[['age']])

embarked_imputer = SimpleImputer(strategy='most_frequent')
df['embarked'] = embarked_imputer.fit_transform(df[['embarked']]).ravel()

# 3. Feature Engineering
df['FamilySize'] = df['sibsp'] + df['parch'] + 1
df['IsAlone'] = np.where(df['FamilySize'] == 1, 1, 0)
df.drop(columns=['sibsp', 'parch'], inplace=True)

# 4. Categorical Encoding
df = pd.get_dummies(df, columns=['sex', 'embarked'], drop_first=True, dtype=int)
for col in ['adult_male', 'alone']:
    df[col] = df[col].astype(int)

# 5. Partitioning: Features (X) and Target (y)
X = df.drop(columns=['survived'])
y = df['survived']

X_train, X_test, y_train, y_test = train_test_split(
    X, y, test_size=0.20, random_state=42, stratify=y
)

# 6. Feature Scaling (Fit on Train, Transform on Test)
scale_cols = ['age', 'fare', 'FamilySize']
scaler = StandardScaler()

X_train_scaled = X_train.copy()
X_test_scaled = X_test.copy()

X_train_scaled[scale_cols] = scaler.fit_transform(X_train[scale_cols])
X_test_scaled[scale_cols] = scaler.transform(X_test[scale_cols])

# 7. Model Training and Performance Evaluation

# Model A: Logistic Regression
lr = LogisticRegression(max_iter=500, random_state=42)
lr.fit(X_train_scaled, y_train)
y_pred_lr = lr.predict(X_test_scaled)

# Model B: K-Nearest Neighbors
knn = KNeighborsClassifier(n_neighbors=5)
knn.fit(X_train_scaled, y_train)
y_pred_knn = knn.predict(X_test_scaled)

# Print Performance Reports
print(f"Logistic Regression Accuracy: {accuracy_score(y_test, y_pred_lr):.4f}")
print("Logistic Regression Report:\n", classification_report(y_test, y_pred_lr))

print(f"\nKNN (k=5) Accuracy:          {accuracy_score(y_test, y_pred_knn):.4f}")
print("KNN Classification Report:\n", classification_report(y_test, y_pred_knn))
```
