```python
# students.py
import random
import datetime

def generate_random_birthdate(start_year: int = 2000, end_year: int = 2006) -> datetime.date:
    """Generates a random birth date between start_year and end_year."""
    start_date = datetime.date(start_year, 1, 1)
    end_date = datetime.date(end_year, 12, 31)
    days_between = (end_date - start_date).days
    random_days = random.randint(0, days_between)
    return start_date + datetime.timedelta(days=random_days)

def generate_student_cohort(num_students: int = 10, seed: int = 42) -> list[dict]:
    """Generates synthetic student records with birth dates and grades."""
    random.seed(seed)
    cohort = []
    for i in range(1, num_students + 1):
        birthdate = generate_random_birthdate(2000, 2006)
        grades = [round(random.uniform(0.0, 20.0), 2) for _ in range(3)]
        student_id = f"STU_{i:03d}"
        cohort.append({
            "student_id": student_id,
            "birthdate": birthdate,
            "grades": grades
        })
    return cohort
```

### Module 2: `analysis.py`
```python
# analysis.py
import datetime
import statistics

def calculate_student_age(birthdate: datetime.date, reference_year: int = 2026) -> int:
    """Calculates age in years relative to a reference year."""
    return reference_year - birthdate.year

def enrich_cohort_data(cohort: list[dict], reference_year: int = 2026) -> list[dict]:
    """Enriches student records with calculated age and grade average."""
    for student in cohort:
        student["age"] = calculate_student_age(student["birthdate"], reference_year)
        student["average_grade"] = round(statistics.mean(student["grades"]), 2)
    return cohort

def compute_class_statistics(cohort: list[dict]) -> dict:
    """Computes descriptive statistics across the student cohort."""
    averages = [s["average_grade"] for s in cohort]
    ages = [s["age"] for s in cohort]

    return {
        "mean_grade": round(statistics.mean(averages), 2),
        "median_grade": round(statistics.median(averages), 2),
        "stdev_grade": round(statistics.stdev(averages), 2),
        "min_grade": round(min(averages), 2),
        "max_grade": round(max(averages), 2),
        "mean_age": round(statistics.mean(ages), 1)
    }
```

### Script / Notebook: `main.ipynb`
```python
import os
from students import generate_student_cohort
from analysis import enrich_cohort_data, compute_class_statistics

# 1. Generate and enrich student cohort
raw_cohort = generate_student_cohort(num_students=10, seed=101)
cohort = enrich_cohort_data(raw_cohort, reference_year=2026)

# 2. Compute descriptive statistics
stats = compute_class_statistics(cohort)

# 3. Create results directory if it doesn't exist
os.makedirs("results", exist_ok=True)
report_path = os.path.join("results", "report.txt")

# 4. Generate and save the text summary
report_content = f"""==================================================
           ACADEMIC PERFORMANCE SUMMARY
==================================================
Total Students Analyzed: {len(cohort)}
Evaluation Reference Year: 2026

Descriptive Metrics:
--------------------------------------------------
- Mean Grade Average:      {stats['mean_grade']} / 20.0
- Median Grade Average:    {stats['median_grade']} / 20.0
- Grade Standard Deviation:{stats['stdev_grade']}
- Minimum Student Average: {stats['min_grade']} / 20.0
- Maximum Student Average: {stats['max_grade']} / 20.0
- Mean Student Age:        {stats['mean_age']} years
==================================================
Individual Records:
"""

for s in cohort:
    report_content += f"{s['student_id']} | Age: {s['age']} | Grades: {s['grades']} | Average: {s['average_grade']}\n"

with open(report_path, "w", encoding="utf-8") as f:
    f.write(report_content)

print(report_content)
print(f"Report successfully saved to: {report_path}")
```
