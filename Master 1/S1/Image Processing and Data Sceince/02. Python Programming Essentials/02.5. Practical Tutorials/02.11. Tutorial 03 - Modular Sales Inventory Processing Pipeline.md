## 3. Implementation of Utility Package

### Step 1: `utils/helpers.py`
```python
# utils/helpers.py

def clean_lines_loop(lines: list[str]) -> list[str]:
    """Cleans a list of lines using traditional loops."""
    cleaned = []
    for line in lines:
        stripped = line.strip()
        if stripped:  # Filter out empty lines
            cleaned.append(stripped)
    return cleaned

def clean_lines(lines: list[str]) -> list[str]:
    """Cleans lines using a concise list comprehension."""
    return [stripped for line in lines if (stripped := line.strip())]

def parse_products(lines: list[str]) -> list[tuple[str, float, int]]:
    """
    Parses cleaned string lines into structured tuples: (name, price, quantity).
    Skips malformed records gracefully.
    """
    parsed = []
    for line in lines:
        parts = line.split(",")
        if len(parts) != 3:
            print(f"[Warning] Invalid field count in record: '{line}'")
            continue
        try:
            name = parts[0].strip()
            price = float(parts[1].strip())
            qty = int(parts[2].strip())
            parsed.append((name, price, qty))
        except ValueError as e:
            print(f"[Warning] Numeric parsing error in row '{line}': {e}")
            continue
    return parsed
```

### Step 2: `utils/product.py`
```python
# utils/product.py

class Product:
    """Represents a product item in an inventory."""
    def __init__(self, name: str, price: float, quantity: int):
        self.name = name
        self.price = price
        self.quantity = quantity

    def total_value(self) -> float:
        """Computes the total inventory value of this product."""
        return self.price * self.quantity

    def __str__(self) -> str:
        return f"Product({self.name}, Price={self.price:.2f}, Qty={self.quantity})"
```

## 4. Pipeline Execution Script: `scripts/create_products.py`

```python
# scripts/create_products.py
import os
import sys

# Append project root to path for package imports
sys.path.append(os.path.abspath(os.path.join(os.path.dirname(__file__), "..")))

from utils import clean_lines, parse_products, Product

def main():
    input_path = os.path.join("data", "raw", "data.txt")
    
    if not os.path.exists(input_path):
        print(f"[Error] File not found: {input_path}")
        sys.exit(1)

    with open(input_path, "r", encoding="utf-8") as f:
        raw_lines = f.readlines()

    cleaned = clean_lines(raw_lines)
    records = parse_products(cleaned)

    products: list[Product] = []
    overall_sales = 0.0

    print("\n--- Inventory Valuation ---")
    for name, price, qty in records:
        prod = Product(name, price, qty)
        products.append(prod)
        val = prod.total_value()
        overall_sales += val
        print(f"{prod.name}: {val:.1f}")

    print("---------------------------")
    print(f"Total Sales: {overall_sales:.1f}")

if __name__ == "__main__":
    main()
```

## 5. Visualization Notebook (`notebooks/analysis.ipynb`)
```python
import os
import sys
import matplotlib.pyplot as plt

sys.path.append("..")
from utils import clean_lines, parse_products, Product

with open("../data/raw/data.txt", "r") as f:
    records = parse_products(clean_lines(f.readlines()))

products = [Product(name, price, qty) for name, price, qty in records]
names = [p.name for p in products]
values = [p.total_value() for p in products]

fig, ax = plt.subplots(figsize=(8, 4.5))
bars = ax.bar(names, values, color="steelblue", edgecolor="black", alpha=0.85)

ax.set_title("Total Valuation by Product Line", fontsize=13, fontweight="bold")
ax.set_xlabel("Product Name", fontsize=11)
ax.set_ylabel("Total Inventory Value ($)", fontsize=11)
ax.grid(axis="y", linestyle="--", alpha=0.5)

# Add value labels above bars
for bar in bars:
    yval = bar.get_height()
    ax.text(bar.get_x() + bar.get_width()/2, yval + 50, f"${yval:.0f}", ha="center", va="bottom", fontsize=10)

plt.tight_layout()
plt.show()
```
