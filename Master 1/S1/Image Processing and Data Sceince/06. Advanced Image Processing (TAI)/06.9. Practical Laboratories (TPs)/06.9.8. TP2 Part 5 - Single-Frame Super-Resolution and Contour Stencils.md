# 06.9.8. TP2 Part 5 - Single-Frame Super-Resolution and Contour Stencils

## 1. Overview
This note covers 2x and 4x single-frame super-resolution on historical document datasets (`Images_TP2_EXO6_SR`), comparing linear interpolations with Pascal Getreuer's edge-adaptive Contour Stencils algorithm, covering Exercise 6 of TP2.

## 2. Super-Resolution Benchmarking Pipeline

```python
import cv2
import numpy as np

def calculate_psnr(orig: np.ndarray, proc: np.ndarray) -> float:
    mse = np.mean((orig.astype(np.float64) - proc.astype(np.float64)) ** 2)
    if mse == 0:
        return float('inf')
    return 10.0 * np.log10((255.0 ** 2) / mse)

def contour_stencil_2x(im: np.ndarray) -> np.ndarray:
    """Factor-2 edge-adaptive super-resolution using Contour Stencils."""
    h, w = im.shape
    out_h, out_w = 2 * h - 1, 2 * w - 1
    out = np.zeros((out_h, out_w), dtype=np.float64)

    # 1. Copy known LR pixels to even coordinates
    for y in range(h):
        for x in range(w):
            out[2 * y, 2 * x] = im[y, x]

    # 2. Interpolate cell centers (X) along path of minimum Total Variation
    for y in range(h - 1):
        for x in range(w - 1):
            v00 = float(im[y, x])
            v01 = float(im[y, x + 1])
            v10 = float(im[y + 1, x])
            v11 = float(im[y + 1, x + 1])

            tv1 = abs(v00 - v11)
            tv2 = abs(v10 - v01)

            if tv1 < tv2:
                x_val = 0.5 * (v00 + v11)
            elif tv2 < tv1:
                x_val = 0.5 * (v10 + v01)
            else:
                x_val = 0.25 * (v00 + v01 + v10 + v11)

            out[2 * y + 1, 2 * x + 1] = x_val

    # 3. Interpolate shared edge midpoints
    for y in range(out_h):
        for x in range(out_w):
            if y % 2 == 1 and x % 2 == 0:
                vals = [out[y - 1, x]] if y > 0 else []
                if y < out_h - 1: vals.append(out[y + 1, x])
                out[y, x] = np.mean(vals)
            elif y % 2 == 0 and x % 2 == 1:
                vals = [out[y, x - 1]] if x > 0 else []
                if x < out_w - 1: vals.append(out[y, x + 1])
                out[y, x] = np.mean(vals)

    return np.clip(np.round(out), 0, 255).astype(np.uint8)

# Load ground truth HR and low-resolution inputs
hr_gt = cv2.imread("Images/Images_TP2_EXO6_SR/HR_GT_512x512/HR_GT_1.jpg", cv2.IMREAD_GRAYSCALE)
lr_256 = cv2.imread("Images/Images_TP2_EXO6_SR/LR_256x256/LR_256_1.jpg", cv2.IMREAD_GRAYSCALE)

if hr_gt is not None and lr_256 is not None:
    # 2x Super-Resolution using standard interpolations
    sr_nearest = cv2.resize(lr_256, (511, 511), interpolation=cv2.INTER_NEAREST)
    sr_bilinear = cv2.resize(lr_256, (511, 511), interpolation=cv2.INTER_LINEAR)
    sr_bicubic = cv2.resize(lr_256, (511, 511), interpolation=cv2.INTER_CUBIC)
    
    # 2x Super-Resolution using Contour Stencils
    sr_stencils = contour_stencil_2x(lr_256)

    gt_target = hr_gt[:511, :511]

    print("--- 2x Super-Resolution PSNR Scores ---")
    print(f"Nearest Neighbor: {calculate_psnr(gt_target, sr_nearest):.2f} dB")
    print(f"Bilinear:         {calculate_psnr(gt_target, sr_bilinear):.2f} dB")
    print(f"Bicubic:          {calculate_psnr(gt_target, sr_bicubic):.2f} dB")
    print(f"Contour Stencils: {calculate_psnr(gt_target, sr_stencils):.2f} dB")
```

## 3. Results Analysis
- **Nearest Neighbor**: Fastest, but introduces severe pixelation and block artifacts.
- **Bilinear**: Produces soft, blurred edge boundaries.
- **Bicubic**: Balances sharpness and smoothness, achieving competitive numerical PSNR scores.
- **Contour Stencils**: Achieves superior subjective edge sharpness on directional strokes, such as historic typography and text characters.
