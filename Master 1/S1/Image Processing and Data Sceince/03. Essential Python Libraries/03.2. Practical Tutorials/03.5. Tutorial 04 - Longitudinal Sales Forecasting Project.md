## 3. Code Implementation and Explanations

### Step 1: `utils.py`
```python
# utils.py
import numpy as np

def generate_random_sales(min_val: int, max_val: int, size: int) -> np.ndarray:
    """Generates an array of random integer sales units within specified bounds."""
    return np.random.randint(low=min_val, high=max_val + 1, size=size)
```

### Step 2: Main Processing Pipeline
```python
import os
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import seaborn as sns
from utils import generate_random_sales

# 1. Data Generation
np.random.seed(42)
os.makedirs("data", exist_ok=True)

# Generate monthly date index for 2025
months = pd.date_range(start="2025-01-01", end="2025-12-01", freq="MS")
num_months = len(months)

df_initial = pd.DataFrame({
    "Date": months,
    "Product_A": generate_random_sales(50, 100, num_months),
    "Product_B": generate_random_sales(30, 80, num_months),
    "Product_C": generate_random_sales(20, 60, num_months),
    "Product_D": generate_random_sales(10, 50, num_months)
})
df_initial.to_csv("data/initial.csv", index=False)

# 2. Metric Calculations
df_final = df_initial.copy()
df_final["Month"] = df_final["Date"].dt.strftime("%b")
product_cols = ["Product_A", "Product_B", "Product_C", "Product_D"]

# Compute monthly sums, averages, and month-over-month growth
df_final["Total_Sales"] = df_final[product_cols].sum(axis=1)
df_final["Average_Sales"] = df_final[product_cols].mean(axis=1)
df_final["MoM_Growth_Pct"] = df_final["Total_Sales"].pct_change() * 100

# Map quarters
def assign_quarter(date) -> str:
    m = date.month
    if m <= 3:
        return "Q1"
    elif m <= 6:
        return "Q2"
    elif m <= 9:
        return "Q3"
    else:
        return "Q4"

df_final["Quarter"] = df_final["Date"].apply(assign_quarter)

# Identify best and worst selling products per month
df_final["Max_Sales_Product"] = df_final[product_cols].idxmax(axis=1)
df_final["Min_Sales_Product"] = df_final[product_cols].idxmin(axis=1)

df_final.to_csv("data/final.csv", index=False)

# 3. Pivot Tables
pivot_quarterly = df_final.pivot_table(
    index="Quarter", 
    values=product_cols + ["Total_Sales"], 
    aggfunc="mean"
)
pivot_quarterly.to_csv("data/output.csv")

# 4. Operational Insights
best_month_row = df_final.loc[df_final["Total_Sales"].idxmax()]
best_product = df_final[product_cols].sum().idxmax()
best_quarter = df_final.groupby("Quarter")["Total_Sales"].sum().idxmax()

print("--- Key Operational Insights ---")
print(f"Top Performing Month:   {best_month_row['Month']} ({best_month_row['Total_Sales']} units)")
print(f"Top Performing Product: {best_product}")
print(f"Top Performing Quarter: {best_quarter}")

# 5. Visualizations
fig, axs = plt.subplots(2, 2, figsize=(15, 10))

# Subplot 1: Monthly Product Trends (Line Chart)
for col in product_cols:
    axs[0, 0].plot(df_final["Month"], df_final[col], marker="o", label=col)
axs[0, 0].set_title("Product Sales Trends", fontweight="bold")
axs[0, 0].legend()

# Subplot 2: Monthly Total Sales (Stacked Bar Chart)
df_final.set_index("Month")[product_cols].plot(kind="bar", stacked=True, ax=axs[0, 1], colormap="viridis")
axs[0, 1].set_title("Monthly Total Sales Volume", fontweight="bold")

# Subplot 3: Sales Density Heatmap
sns.heatmap(df_final.set_index("Month")[product_cols], annot=True, fmt="d", cmap="YlGnBu", ax=axs[1, 0])
axs[1, 0].set_title("Sales Density Heatmap", fontweight="bold")

# Subplot 4: Product Variance Boxplots
sns.boxplot(data=df_final[product_cols], ax=axs[1, 1], palette="Set2")
axs[1, 1].set_title("Product Sales Distributions", fontweight="bold")

plt.tight_layout()
plt.show()
```
