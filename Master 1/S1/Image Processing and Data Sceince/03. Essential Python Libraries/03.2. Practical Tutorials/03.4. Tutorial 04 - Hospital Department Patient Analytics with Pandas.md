## 2. Ingestion and Index Manipulation (Exercise 03)

```python
import pandas as pd
import numpy as np

# 1. Instantiate the raw clinical DataFrame
data = {
    'City': ['Alger', 'Annaba', 'Oran', 'Alger', 'Oran', 'Annaba', 'Alger', 'Oran'],
    'Department': ['Cardiology', 'Neurology', 'Orthopedics', 'Cardiology', 'Neurology', 'Orthopedics', 'Cardiology', 'Neurology'],
    'Age': [45, 60, 30, 50, 40, 70, 35, 55],
    'DaysAdmitted': [5, 8, 3, 7, 4, 6, 2, 9],
    'DailyCost': [200, 300, 150, 220, 280, 160, 210, 290],
    'Satisfaction': [4.5, 3.8, 4.2, 4.0, 4.1, 3.5, 4.7, 3.9],
    'Readmitted': [0, 1, 0, 0, 1, 1, 0, 1]
}
df = pd.DataFrame(data)

# 2. Save original DataFrame to disk preserving its default index
df.to_csv("hospital_original.csv", index=True)

# 3. Replace integer index with custom labels 'A' through 'H'
df.index = ['A', 'B', 'C', 'D', 'E', 'F', 'G', 'H']
print("DataFrame with Custom Label Index:")
display(df)

# 4. Slice patients using labels from 'C' to 'F' (inclusive)
display(df.loc['C':'F'])

# 5. Select records at integer positions 1, 2, and 3
display(df.iloc[1:4])

# 6. Select specific columns across label range 'B' to 'F'
display(df.loc['B':'F', ['City', 'Department', 'DailyCost']])

# 7. Filter: Patients older than 45 with Satisfaction below 4.0
cond_filter = (df['Age'] > 45) & (df['Satisfaction'] < 4.0)
display(df.loc[cond_filter, ['City', 'Age', 'Satisfaction', 'Readmitted']])

# 8. Create neurology subset sorted by DailyCost
df_neurology = df[df['Department'] == 'Neurology'][['City', 'Age', 'DaysAdmitted', 'DailyCost']].reset_index(drop=True)
df_neurology = df_neurology.set_index('City').sort_values(by='DailyCost', ascending=False)
display(df_neurology)

# 9. Patient count by city within neurology department
city_counts = df_neurology.index.value_counts()
print("\nNeurology Patients by City:\n", city_counts)
```

## 3. Data Transformations, Risk Scores, and Pivot Tables (Exercise 04)

```python
# 1. Feature Engineering: Total Cost and Age Groups
df['TotalCost'] = df['DaysAdmitted'] * df['DailyCost']

def categorize_age(age: int) -> str:
    if age < 30:
        return "Young"
    elif 30 <= age <= 50:
        return "Middle"
    else:
        return "Senior"

df['AgeGroup'] = df['Age'].apply(categorize_age)

# 2. Modify and Rename Columns
df['Bonus'] = df['DailyCost'] * 0.10
df.rename(columns={'DailyCost': 'DailyFee'}, inplace=True)
df.drop(columns=['Bonus'], inplace=True)

# 3. Classifications: Cost and Satisfaction
# Flag high cost treatments (threshold: > 1200)
df['HighCost'] = np.where(df['TotalCost'] > 1200, "Yes", "No")

def categorize_satisfaction(score: float) -> str:
    if score >= 4.5:
        return "Excellent"
    elif 3.5 <= score < 4.5:
        return "Average"
    else:
        return "Poor"

df['SatisfactionLevel'] = df['Satisfaction'].apply(categorize_satisfaction)

# 4. Ranking and Risk Score Computation
df['CostRank'] = df['TotalCost'].rank(ascending=False, method='min')

# Risk Score Formula:
# RiskScore = (DaysAdmitted * 0.4) + (Age * 0.3) + (100 - Satisfaction * 10) * 0.3
df['RiskScore'] = (df['DaysAdmitted'] * 0.4) + (df['Age'] * 0.3) + (100 - df['Satisfaction'] * 10) * 0.3

def categorize_risk(score: float) -> str:
    if score > 60:
        return "High"
    elif 40 <= score <= 60:
        return "Medium"
    else:
        return "Low"

df['RiskCategory'] = df['RiskScore'].apply(categorize_risk)
df.sort_values(by='RiskScore', ascending=False, inplace=True)

# 5. City Summaries
city_summary = df.groupby('City').agg(
    AvgTotalCost=('TotalCost', 'mean'),
    AvgSatisfaction=('Satisfaction', 'mean'),
    ReadmissionRate=('Readmitted', 'mean')
).reset_index()

highest_risk_city = df.groupby('City')['RiskScore'].mean().idxmax()
print(f"City with Highest Average Clinical Risk: {highest_risk_city}")

# 6. Data Cleaning and Standardization
# Impute missing satisfaction scores with column mean
df['Satisfaction'] = df['Satisfaction'].fillna(df['Satisfaction'].mean())

# Map regional codes and binary flags
region_codes = {'Alger': 'DZ-16', 'Annaba': 'DZ-23', 'Oran': 'DZ-31'}
df['RegionCode'] = df['City'].map(region_codes)
df['ReadmissionStatus'] = df['Readmitted'].map({0: "No", 1: "Yes"})

# 7. Reshaping with Pivot Tables
pivot_results = df.pivot_table(
    index='City', 
    columns='Department', 
    values=['TotalCost', 'Satisfaction'], 
    aggfunc='mean'
)

# 8. Export Processed Datasets
df.to_csv("hospital_risk.csv", index=False)
city_summary.to_csv("city_summary.csv", index=False)

print("\n--- Final Processed Cohort ---")
display(df[['City', 'RegionCode', 'TotalCost', 'RiskScore', 'RiskCategory', 'ReadmissionStatus']])
```
