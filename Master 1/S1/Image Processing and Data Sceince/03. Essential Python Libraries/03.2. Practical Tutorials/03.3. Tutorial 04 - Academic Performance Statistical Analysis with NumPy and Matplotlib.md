# 03.3. Tutorial 04 - Academic Performance Statistical Analysis with NumPy and Matplotlib

## 1. Overview
This tutorial analyzes an academic dataset of 100 students across 5 modules (`THL`, `DSS`, `DB`, `SE`, `RE`) using NumPy and Matplotlib. We evaluate descriptive metrics, percentiles, boolean masks, and distributions.

## 2. Code Implementation and Explanations

```python
import numpy as np
import matplotlib.pyplot as plt

# 1. Dataset Initialization
np.random.seed(101)  # Fix seed for reproducible grades
# Shape: (100 students, 5 modules); Grades in range [0, 20]
grades_of_students = np.random.randint(0, 21, size=(100, 5))
modules = ['THL', 'DSS', 'DB', 'SE', 'RE']

print("--- Preview of First 5 Student Grade Vectors ---")
print(grades_of_students[:5])

# 2. Descriptive Statistical Metrics
mean_per_student = grades_of_students.mean(axis=1)
mean_per_module = grades_of_students.mean(axis=0)
median_per_module = np.median(grades_of_students, axis=0)
var_per_module = grades_of_students.var(axis=0)
std_per_module = grades_of_students.std(axis=0)

print("\n--- Module Performance Summaries ---")
for idx, mod in enumerate(modules):
    print(f"Module {mod:4s}: Mean={mean_per_module[idx]:.2f} | Median={median_per_module[idx]:.1f} | Variance={var_per_module[idx]:.2f} | StdDev={std_per_module[idx]:.2f}")

print(f"\nOverall Minimum Grade: {grades_of_students.min()}")
print(f"Overall Maximum Grade: {grades_of_students.max()}")

# 3. Percentiles for THL (Column Index 0)
thl_grades = grades_of_students[:, 0]
p25, p50, p75 = np.percentile(thl_grades, [25, 50, 75])
print(f"\nTHL Module Percentiles: 25th={p25:.1f}, 50th={p50:.1f}, 75th={p75:.1f}")

# 4. Boolean Masking: Grades between 10 and 15 (inclusive)
passing_mask = (grades_of_students >= 10) & (grades_of_students <= 15)
filtered_grades = grades_of_students[passing_mask]
print(f"\nTotal grades between 10 and 15: {filtered_grades.size} out of {grades_of_students.size}")

# 5. Visualizations
fig, axs = plt.subplots(1, 2, figsize=(14, 5))

# Plot A: Average Grade Comparison
axs[0].bar(modules, mean_per_module, color="skyblue", edgecolor="black", alpha=0.85)
axs[0].axhline(10.0, color="red", linestyle="--", label="Passing Threshold (10/20)")
axs[0].set_title("Module Average Grade Comparison", fontsize=12, fontweight="bold")
axs[0].set_ylabel("Average Score (/20)")
axs[0].set_ylim(0, 20)
axs[0].legend()
axs[0].grid(axis="y", linestyle=":", alpha=0.6)

# Plot B: Grade Distributions (Boxplot)
axs[1].boxplot(grades_of_students, labels=modules, patch_artist=True,
               boxprops=dict(facecolor="lightgreen", color="black"),
               medianprops=dict(color="darkred", linewidth=2.0))
axs[1].set_title("Grade Distributions Across Modules", fontsize=12, fontweight="bold")
axs[1].set_ylabel("Grade (/20)")
axs[1].grid(axis="y", linestyle=":", alpha=0.6)

plt.tight_layout()
plt.show()

# 6. Extra Practice: Cohort Benchmarking
# Identify students with all grades >= 15
high_achievers = np.where((grades_of_students >= 15).all(axis=1))[0]
print(f"\nStudents scoring >= 15 across all modules: {high_achievers.tolist()}")

# Identify top 5 students by overall average
top5_indices = np.argsort(mean_per_student)[::-1][:5]
print("Top 5 Students by Grade Average:")
for rank, idx in enumerate(top5_indices, start=1):
    print(f"  Rank {rank}: Student #{idx:02d} - Average: {mean_per_student[idx]:.2f}")
```
